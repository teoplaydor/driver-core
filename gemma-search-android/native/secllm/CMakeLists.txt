# The secretary's local model: llama.cpp (PrismML's fork — their ternary and 1-bit kernels) + secllm.cpp, one .so.
cmake_minimum_required(VERSION 3.22)
project(secllm CXX C)

set(LLAMA_DIR "" CACHE PATH "llama.cpp sources")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_DIR} llama EXCLUDE_FROM_ALL)

if (NOT ANDROID)
  find_package(JNI REQUIRED)
endif()
add_library(secllm SHARED secllm.cpp)
target_link_libraries(secllm PRIVATE llama)
if (NOT ANDROID)
  target_include_directories(secllm PRIVATE ${JNI_INCLUDE_DIRS})
else()
  target_link_libraries(secllm PRIVATE log)
  # only the JNI entry points are exported: the linker drops what they do not reach
  target_link_options(secllm PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
endif()
set_target_properties(secllm PROPERTIES CXX_STANDARD 17)
